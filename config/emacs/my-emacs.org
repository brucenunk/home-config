#+title: Emacs configuration
#+startup: overview
#+property: header-args:emacs-lisp :tangle no :mkdirp yes :comments no :results silent

This is the authoring entry point for our Emacs configuration, inspired by
[[https://protesilaos.com/emacs/dotemacs][Prot's literate configuration]]. It records what is enabled and why. The
implementation lives in [[file:my-lisp/][my-lisp/]]; tests live in [[file:tests/][tests/]].

Nix tangles this document during the build and deploys ordinary Lisp. Emacs
never parses this document at startup. Do not edit the generated files.
Use =scripts/tangle-emacs-config OUTPUT-DIRECTORY= for a disposable local
render; see [[file:../../docs/emacs-literate.md][the maintenance guide]] for verification and deployment boundaries.

Nix owns package and tree-sitter grammar installation. =use-package= only
configures those packages. Each package has one owning declaration. Hooks,
key bindings, settings and integration calls stay here; function bodies stay
in handwritten libraries. Unknown rationale is marked with a =# TODO:= comment.

* Early startup
:PROPERTIES:
:header-args:emacs-lisp: :tangle early-init.el :mkdirp yes :comments no
:END:

Avoid startup chrome and keep package provisioning in Nix. Package activation
remains enabled so Nix-provided autoloads are available. The Custom file is a
local override, not a second repository-maintained configuration source.

;; -*- lexical-binding: t -*-
;; Generated from my-emacs.org; do not edit.

(setq-default backup-inhibited t
	      create-lockfiles nil
              custom-file (expand-file-name "custom.el" user-emacs-directory)
	      inhibit-startup-message t
	      make-backup-files nil
	      native-comp-async-report-warnings-errors 'silent
	      use-short-answers t)

;; Keep package.el startup activation for Nix-provided package autoloads, but
;; leave package provisioning exclusively to Nix.
(setq package-archives nil)

(menu-bar-mode -1)
(scroll-bar-mode -1)
(tool-bar-mode -1)

(when (featurep 'ns)
  (setq ns-use-thin-smoothing t)
  (add-to-list 'default-frame-alist '(ns-transparent-titlebar . t)))

(setq frame-resize-pixelwise t)
#+end_src

** Prose with fixed-width code

Org, Markdown and agent viewports use mixed-pitch text. Keep code and tables
fixed-width, including renderer-specific faces. Refresh remaps after font changes.

#+begin_src emacs-lisp
(use-package mixed-pitch
  :ensure nil
  :hook ((markdown-mode . mixed-pitch-mode)
         (org-mode . mixed-pitch-mode)
         (agent-shell-viewport-view-mode . mixed-pitch-mode)
         (agent-shell-viewport-edit-mode . mixed-pitch-mode))
  :config
  ;; mixed-pitch already keeps Org blocks, code/verbatim, tables, metadata,
  ;; checkboxes, and indentation fixed-pitch; retain its native face list.
  ;; The renderer's base faces also cover source labels, syntax-highlighted
  ;; bodies, and table headers/borders through inheritance.  Remap them only
  ;; in mixed-pitch buffers; ordinary shell buffers keep their own font.
  ;; The composer reuses Org or Markdown faces, keeping all markup editable.
  ;; Include pre-face for untyped/unknown-language code bodies too.
  (dolist (face '(markdown-table-face
                  markdown-pre-face
                  agent-shell-markdown-inline-code
                  agent-shell-markdown-source-block
                  agent-shell-markdown-table))
    (add-to-list 'mixed-pitch-fixed-pitch-faces face))
  (add-hook 'mixed-pitch-mode-hook #'my/viewport-prompt-variable-pitch)
  (add-hook 'fontaine-set-preset-hook #'my/refresh-mixed-pitch-mode))
#+end_src

** Explicit file and word completion

Automatic file completion only handles explicit paths, leaving prose, URLs,
mentions and slash commands alone. =C-c p f= offers manual file completion;
a prefix argument chooses a temporary base directory. Dabbrev starts at three
characters and follows file completion in the fallback chain.

#+begin_src emacs-lisp
(use-package cape

  :config

Use Consult previews for buffers, search, diagnostics and xref locations.

(use-package consult
  :ensure nil
  :bind
  (;; C-c bindings in `mode-specific-map'
   ("C-c i" . consult-info)
   ;; C-x bindings in `ctl-x-map'
   ("C-x b" . consult-buffer)
   ;; M-g bindings in `goto-map'
   ("M-g f" . consult-flymake)
   ;; M-s bindings in `search-map'

   ("M-s g" . consult-ripgrep)
   ("M-s l" . consult-line)

  :demand t
  :config

  (declare-function consult-xref "consult-xref")

        xref-show-definitions-function #'consult-xref)
  (consult-customize
   consult-line
   :add-history (seq-some #'thing-at-point '(region symbol)))

  (defalias 'consult-line-thing-at-point 'consult-line)

  (consult-customize
   consult-line-thing-at-point

=tasks/= subtree. Show notes on empty startup without replacing explicitly

#+begin_src emacs-lisp
(use-package denote
  :ensure nil
  :init
  (add-hook 'emacs-startup-hook #'my/denote-startup)
  :demand t
  :hook ((dired-mode    . denote-dired-mode-in-directories)
         (markdown-mode . denote-fontify-links-mode-maybe))
  :bind (("C-c n T" . my/denote-task)
         ("C-c n n" . denote)
         ("C-c n t" . denote-template)
         ("C-c n r" . denote-rename-file)
         ("C-c n l" . denote-link)
         ("C-c n b" . denote-backlinks))
  :config
  (require 'my-denote-task)
  (setq denote-templates
        '((task-workflow . (lambda () (my/denote-template "task-workflow.org")))))
  (setq denote-directory (expand-file-name "~/work/docs/"))
  (setq denote-dired-directories (list denote-directory))
  (setq denote-dired-directories-include-subdirectories t)
  (setq denote-known-keywords nil)
  (setq denote-file-type 'org)
  (setq denote-rename-confirmations nil)

  (advice-add #'denote-link :around #'my/denote-link-normalize-file-arg)
  (add-to-list 'revert-without-query
               (concat "^" (regexp-quote (file-name-as-directory denote-directory))))
  (denote-rename-buffer-mode 1)
  (remove-hook 'denote-after-new-note-hook
               #'denote-rename-buffer-rename-function-or-fallback)
  (remove-hook 'find-file-hook #'denote-rename-buffer-rename-function-or-fallback)
  (add-hook 'denote-after-new-note-hook #'my/denote-rename-new-buffer-maybe)

Only start Eglot for allowlisted modes when their configured executable is

(use-package eglot
  :ensure nil
  :preface
  ;; Keep this allowlist in sync with language servers installed by Nix.
  ;; Avoid a broad `prog-mode' hook: it makes Eglot noisy for modes without
  ;; a configured/available server.
  (setq my/eglot-mode-server-executables
    '((bash-ts-mode . "bash-language-server")
      (go-ts-mode . "gopls")
      (jsonnet-mode . "jsonnet-language-server")
      (nix-ts-mode . "nixd")
      (python-ts-mode . "pylsp")
      (rego-mode . "regal")
      (terraform-mode . "terraform-ls")
      (yaml-mode . "yaml-language-server")))

  :hook
  (eglot-managed-mode . my/eglot-before-save-hook)
  ((bash-ts-mode go-ts-mode jsonnet-mode nix-ts-mode python-ts-mode
    rego-mode terraform-mode yaml-mode) . my/eglot-ensure-if-server-available)
  :custom
  (eglot-events-buffer-config '(:size 0) "Disable the events buffer")
  (eglot-connect-timeout 120 "Longer for servers like `gopls' to boot over TRAMP")
  (eglot-confirm-server-initiated-edits nil "Don't ask for confirmation")
  (eglot-autoshutdown t "Shutdown server when last managed buffer is killed")
  (eglot-sync-connect nil "Never block LSP connection attempts")
  (eglot-send-changes-idle-time 0.2 "Send data to the server faster")
  :config
  (add-to-list 'eglot-server-programs
               '(nix-ts-mode . ("nixd" :initializationOptions
                                (:formatting (:command ["nixfmt"]))))))
